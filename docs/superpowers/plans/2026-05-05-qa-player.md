> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a lightweight standalone Q&A player for `umto_symbolic_meter` exercises that fixes blank render and broken navigation by bypassing the RequireJS/AMD/ExerciseContext stack entirely.

**Architecture:** Django template branch routes `umto_symbolic_meter` exercises to a new `qa_player.html` template that bootstraps `qa_player.js` — a plain ES module (~250 lines) with its own state loop, render functions, and AJAX save. The existing exercise shell (`exercise.html` + AMD app) is untouched.

**Tech Stack:** Vanilla JS (ES modules), Django templates, existing `harmony.css`, existing `/ajax/identification-answers/` endpoint.

**Spec:** `docs/superpowers/specs/2026-05-05-qa-player-design.md`

## Task 2: Django template

**Files:**
- Create: `lab/templates/qa_player.html`

Extends `__base.html`. No RequireJS, no piano, no `#staffwrapper`. Injects config as `window.__EXERCISE_CONFIG__` and loads the ES module player.

`__base.html` has two relevant blocks:
- `{% block content %}` — page body HTML
- `{% block js_extra %}` — scripts after require.js (we don't need require.js at all, but it loads from `{% block js_require %}` in `__base.html` — we can't suppress it, so just load our module in `js_extra`)

- [ ] **Step 1: Create the template**

```html
{% extends "__base.html" %}
{% load static %}

{% block content %}
<div id="qa-container">
  {% include "_header_music.html" %}
  <div class="harmony-ui">
    <div id="staff-area" class="content-area">
      <div id="qa-root"></div>
      <div class="instruction-panel">
        <div class="instruction-panel-header">
          <div><h4>{% if playlist_name %}{{ playlist_name }}{% else %}PREVIEW{% endif %}</h4></div>
        </div>
      </div>
    </div>
  </div>
</div>
{% endblock %}

{% block js_extra %}
<script>
  window.__EXERCISE_CONFIG__ = {{ exercise_config_json }};
</script>
<script type="module" src="{% static 'js/src/qa_player.js' %}"></script>
{% endblock %}
```

Note: `exercise_config_json` is already `json.dumps(...)` output from the view — it is safe HTML (Django's `json.dumps` does not produce `</script>` sequences for normal data). No `|safe` filter needed since we're using `{{ }}` not `{% autoescape off %}`.

Actually Django auto-escapes `{{ }}` — use `{{ exercise_config_json|safe }}` since the view already produced trusted JSON:

```html
  window.__EXERCISE_CONFIG__ = {{ exercise_config_json|safe }};
```

## Task 4: `qa_player.js` — state, rendering, and AJAX

**Files:**
- Create: `lab/static/js/src/qa_player.js`

Full standalone ES module. No RequireJS, no jQuery, no external dependencies. Inlines the time-signature rendering logic (32 lines) to avoid the AMD dependency on `symbolic_meter_rendering_logic.js`.

### Key config fields (from `window.__EXERCISE_CONFIG__`):

```
config.type             // "umto_symbolic_meter"
config.problems         // array of problem objects
config.inputMode        // "flashcard" | "quiz" | undefined → default "flashcard"
config.exerciseId       // string e.g. "EB16IC"
config.exerciseNum      // int
config.playlistName     // string — this is the playlist ID (e.g. "PA01DY")
config.courseId         // string or null
config.nextExercise     // URL string or null/false
```

### Problem object shape:

```
problem.prompt          // { kind: "time_signature"|"meter_label", text: "3/4" }
problem.choices         // ["duple", "triple", "quadruple", ...]
problem.answer          // "triple"
```

- [ ] **Step 1: Create `lab/static/js/src/qa_player.js`**

```javascript
(function () {
  "use strict";

  // ── Time-signature rendering (inlined from symbolic_meter_rendering_logic.js) ──

  var MUSIC_FONT = '"Bravura Text","Finale Maestro Text","Petaluma Script","Noto Music",serif';

  function renderPromptHTML(prompt) {
    if (!prompt) return "";
    if (prompt.kind === "time_signature" && /^\d+\/\d+$/.test(prompt.text)) {
      var parts = prompt.text.split("/");
      return (
        '<span class="meter-ts" style="font-family:' + MUSIC_FONT + '">' +
        '<span class="meter-ts-top">' + parts[0] + "</span>" +
        '<span class="meter-ts-bottom">' + parts[1] + "</span>" +
        "</span>"
      );
    }
    return '<span>' + escapeHTML(prompt.text || "") + '</span>';
  }

  function escapeHTML(str) {
    return String(str)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;");
  }

  // ── CSRF ────────────────────────────────────────────────────────────────────

  function getCookie(name) {
    var value = "; " + document.cookie;
    var parts = value.split("; " + name + "=");
    if (parts.length === 2) return parts.pop().split(";").shift();
    return "";
  }

  // ── Persistence ─────────────────────────────────────────────────────────────

  var STORAGE_TTL_MS = 7 * 24 * 60 * 60 * 1000;

  function loadSaved(exerciseId) {
    try {
      var raw = localStorage.getItem("qa:" + exerciseId);
      if (!raw) return null;
      var obj = JSON.parse(raw);
      if (Date.now() - (obj.savedAt || 0) > STORAGE_TTL_MS) return null;
      return obj;
    } catch (e) {
      return null;
    }
  }

  function saveState(state) {
    try {
      localStorage.setItem(
        "qa:" + state.exerciseId,
        JSON.stringify({
          answers: state.answers,
          revealed: state.revealed,
          page: state.page,
          submitted: state.submitted,
          savedAt: Date.now(),
        })
      );
    } catch (e) {}
  }

  // ── AJAX save ───────────────────────────────────────────────────────────────

  function saveResults(config, answers) {
    var problems = config.problems || [];
    var correctness = {};
    var correct_answers = {};
    var correctCount = 0;
    var incorrectCount = 0;

    problems.forEach(function (p, i) {
      var chosen = answers[i];
      var isCorrect = chosen === p.answer;
      correctness[i] = isCorrect;
      correct_answers[i] = p.answer;
      if (isCorrect) correctCount++;
      else incorrectCount++;
    });

    var payload = {
      course_ID: config.courseId || null,
      playlist_ID: config.playlistName,
      exercise_num: config.exerciseNum,
      answers: answers,
      correctness: correctness,
      score: {
        correct: correctCount,
        incorrect: incorrectCount,
        unanswered: 0,
      },
      correct_answers: correct_answers,
      page_num: 0,
      is_page_submit: false,
      mode: config.inputMode || "flashcard",
      attempt_type: "practice",
    };

    fetch("/ajax/identification-answers/", {
      method: "POST",
      headers: {
        "Content-Type": "application/x-www-form-urlencoded",
        "X-CSRFToken": getCookie("csrftoken"),
      },
      body: "data=" + encodeURIComponent(JSON.stringify(payload)),
    }).catch(function () {});
  }

  // ── State ────────────────────────────────────────────────────────────────────

  function makeInitialState(config) {
    var saved = loadSaved(config.exerciseId);
    return {
      exerciseId: config.exerciseId,
      problems: config.problems || [],
      mode: config.inputMode === "quiz" ? "quiz" : "flashcard",
      answers: (saved && saved.answers) || {},
      revealed: (saved && saved.revealed) || {},
      page: (saved && saved.page) || 0,
      submitted: (saved && saved.submitted) || false,
      nextExerciseUrl: config.nextExercise || null,
    };
  }

  function dispatch(state, action, config) {
    var s = Object.assign({}, state, {
      answers: Object.assign({}, state.answers),
      revealed: Object.assign({}, state.revealed),
    });

    if (action.type === "SELECT_ANSWER") {
      if (s.mode === "flashcard") {
        if (s.revealed[action.index]) return s; // already answered
        s.answers[action.index] = action.answer;
        s.revealed[action.index] = true;
        saveState(s);
        var isCorrect = action.answer === s.problems[action.index].answer;
        if (isCorrect) {
          var isLast = action.index >= s.problems.length - 1;
          if (isLast) {
            saveResults(config, s.answers);
            // Clear saved state on completion
            try { localStorage.removeItem("qa:" + s.exerciseId); } catch(e) {}
            setTimeout(function () {
              if (s.nextExerciseUrl) window.location.href = s.nextExerciseUrl;
            }, 1000);
          } else {
            setTimeout(function () {
              render(
                Object.assign({}, s, { page: action.index + 1 }),
                config
              );
            }, 800);
          }
        }
      } else {
        // quiz mode: just record, no reveal yet
        if (s.submitted) return s;
        s.answers[action.index] = action.answer;
        saveState(s);
      }
    }

    if (action.type === "NEXT_PAGE") {
      if (s.mode === "flashcard") {
        s.page = Math.min(action.page, s.problems.length - 1);
      }
    }

    if (action.type === "SUBMIT_QUIZ") {
      if (s.mode === "quiz" && !s.submitted) {
        s.submitted = true;
        s.revealed = {};
        s.problems.forEach(function (_, i) { s.revealed[i] = true; });
        saveResults(config, s.answers);
        try { localStorage.removeItem("qa:" + s.exerciseId); } catch(e) {}
        saveState(s);
      }
    }

    return s;
  }

  // ── Render ───────────────────────────────────────────────────────────────────

  function render(state, config) {
    var root = document.getElementById("qa-root");
    if (!root) return;
    if (state.mode === "quiz") {
      root.innerHTML = buildQuizHTML(state);
    } else {
      root.innerHTML = buildFlashcardHTML(state);
    }
    bindEvents(root, state, config);
  }

  function buildFlashcardHTML(state) {
    var problems = state.problems;
    var page = Math.min(state.page, problems.length - 1);
    var problem = problems[page];
    if (!problem) return '<p style="padding:24px">No problems found.</p>';

    var chosen = state.answers[page];
    var isRevealed = !!state.revealed[page];
    var isCorrect = isRevealed && chosen === problem.answer;

    var choicesHTML = (problem.choices || []).map(function (choice) {
      var cls = "qa-choice";
      if (isRevealed && chosen === choice) {
        cls += isCorrect ? " correct" : " incorrect";
      }
      var disabled = isRevealed ? " disabled" : "";
      return (
        '<button type="button" class="' + cls + '"' + disabled +
        ' data-choice="' + escapeHTML(choice) + '"' +
        ' data-index="' + page + '">' +
        escapeHTML(choice) + "</button>"
      );
    }).join("");

    var feedbackHTML = "";
    if (isRevealed) {
      if (isCorrect) {
        feedbackHTML = '<div class="qa-feedback qa-feedback-correct">Correct!</div>';
      } else {
        feedbackHTML =
          '<div class="qa-feedback qa-feedback-incorrect">Incorrect — correct answer: ' +
          escapeHTML(problem.answer) + "</div>" +
          '<button type="button" class="qa-next-btn" data-action="next" data-page="' +
          (page + 1) + '">Next</button>';
      }
    }

    return (
      '<div class="qa-progress">' + (page + 1) + " / " + problems.length + "</div>" +
      '<div class="qa-card">' +
      '<div class="qa-prompt">' + renderPromptHTML(problem.prompt) + "</div>" +
      '<div class="qa-choices">' + choicesHTML + "</div>" +
      feedbackHTML +
      "</div>"
    );
  }

  function buildQuizHTML(state) {
    var problems = state.problems;
    var allAnswered = problems.every(function (_, i) {
      return state.answers[i] !== undefined;
    });

    var cardsHTML = problems.map(function (problem, i) {
      var chosen = state.answers[i];
      var isRevealed = !!state.revealed[i];
      var isCorrect = isRevealed && chosen === problem.answer;

      var choicesHTML = (problem.choices || []).map(function (choice) {
        var cls = "qa-choice";
        if (!isRevealed && chosen === choice) cls += " selected";
        if (isRevealed && chosen === choice) {
          cls += isCorrect ? " correct" : " incorrect";
        }
        var disabled = isRevealed ? " disabled" : "";
        return (
          '<button type="button" class="' + cls + '"' + disabled +
          ' data-choice="' + escapeHTML(choice) + '"' +
          ' data-index="' + i + '">' +
          escapeHTML(choice) + "</button>"
        );
      }).join("");

      var feedbackHTML = "";
      if (isRevealed) {
        feedbackHTML = isCorrect
          ? '<div class="qa-feedback qa-feedback-correct">Correct</div>'
          : '<div class="qa-feedback qa-feedback-incorrect">Incorrect — ' +
            escapeHTML(problem.answer) + "</div>";
      }

      return (
        '<div class="qa-card">' +
        '<div class="qa-prompt">' + renderPromptHTML(problem.prompt) + "</div>" +
        '<div class="qa-choices">' + choicesHTML + "</div>" +
        feedbackHTML +
        "</div>"
      );
    }).join("");

    var bottomHTML = "";
    if (state.submitted) {
      var correctCount = problems.filter(function (p, i) {
        return state.answers[i] === p.answer;
      }).length;
      bottomHTML =
        '<div class="qa-complete-banner">Done — ' +
        correctCount + " / " + problems.length + " correct" +
        (state.nextExerciseUrl
          ? '<br><a class="qa-next-exercise-btn" href="' +
            escapeHTML(state.nextExerciseUrl) + '">Next Exercise</a>'
          : "") +
        "</div>";
    } else {
      bottomHTML =
        '<div class="qa-submit-bar">' +
        '<button type="button" class="qa-submit-btn" data-action="submit"' +
        (allAnswered ? "" : " disabled") + ">Submit</button>" +
        "</div>";
    }

    return cardsHTML + bottomHTML;
  }

  function bindEvents(root, state, config) {
    root.addEventListener("click", function (e) {
      var btn = e.target.closest("button");
      if (!btn) return;

      var action = btn.dataset.action;
      var choice = btn.dataset.choice;
      var indexStr = btn.dataset.index;

      if (action === "submit") {
        var newState = dispatch(state, { type: "SUBMIT_QUIZ" }, config);
        render(newState, config);
        return;
      }

      if (action === "next") {
        var nextPage = parseInt(btn.dataset.page, 10);
        var newState = dispatch(state, { type: "NEXT_PAGE", page: nextPage }, config);
        render(newState, config);
        return;
      }

      if (choice !== undefined && indexStr !== undefined) {
        var index = parseInt(indexStr, 10);
        var newState = dispatch(state, { type: "SELECT_ANSWER", index: index, answer: choice }, config);
        render(newState, config);
        return;
      }
    });
  }

  // ── Boot ─────────────────────────────────────────────────────────────────────

  document.addEventListener("DOMContentLoaded", function () {
    var config = window.__EXERCISE_CONFIG__;
    if (!config || config.type !== "umto_symbolic_meter") return;
    var state = makeInitialState(config);
    render(state, config);
  });
})();
```
